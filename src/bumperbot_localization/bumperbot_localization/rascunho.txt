#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
import time
from sensor_msgs.msg import Imu
from nav_msgs.msg import Odometry
import tf2_ros
from geometry_msgs.msg import TransformStamped

class ImuRepublisherNode(Node):
    def __init__(self):
        super().__init__("imu_republisher_node")
        
        # Publisher para mensagens IMU modificadas
        self.imu_pub = self.create_publisher(Imu, "imu_ekf", 10)
        
        # Subscriber para mensagens IMU originais
        self.imu_sub = self.create_subscription(Imu, "imu/out", self.imu_callback, 10)
        
        # Subscriber para odometria
        self.odom_sub = self.create_subscription(Odometry, "bumperbot_controller/odom", self.odom_callback, 10)
        
        # Broadcaster de TF para publicar transformações
        self.tf_broadcaster = tf2_ros.TransformBroadcaster(self)
        
        self.get_logger().info("IMU Republisher iniciado e publicando transformações dinâmicas")

    def imu_callback(self, imu):
        # Republicar o IMU com o frame_id modificado
        imu.header.frame_id = "base_footprint_ekf"
        self.imu_pub.publish(imu)

    def odom_callback(self, msg):
        self.get_logger().info(f"Recebendo mensagem de odometria: {msg.header.stamp}")
        # resto do código...
        t = TransformStamped()
        
        # Preencher cabeçalho
        t.header.stamp = self.get_clock().now().to_msg()
        t.header.frame_id = "odom"
        t.child_frame_id = "base_footprint_ekf"
        
        # Definir a posição da transformação
        t.transform.translation.x = msg.pose.pose.position.x
        t.transform.translation.y = msg.pose.pose.position.y
        t.transform.translation.z = msg.pose.pose.position.z
        
        # Definir orientação (quaternion)
        t.transform.rotation.x = msg.pose.pose.orientation.x
        t.transform.rotation.y = msg.pose.pose.orientation.y
        t.transform.rotation.z = msg.pose.pose.orientation.z
        t.transform.rotation.w = msg.pose.pose.orientation.w
        
        # Publicar a transformação
        self.tf_broadcaster.sendTransform(t)

def main():
    rclpy.init()
    node = ImuRepublisherNode()
    rclpy.spin(node)
    rclpy.shutdown()

if __name__ == "__main__":
    main()
